def load_base(path: str) -> Any:
    spec = importlib.util.spec_from_file_location("frozen_okx_base", path)
    if spec is None or spec.loader is None:
        raise RuntimeError("cannot load frozen acquisition base")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


def corr(x: np.ndarray, y: np.ndarray) -> float:
    xc = x - x.mean()
    yc = y - y.mean()
    denominator = math.sqrt(float(np.sum(xc * xc) * np.sum(yc * yc)))
    return float(np.sum(xc * yc) / denominator) if denominator > 0 else float("nan")


def ranks(x: np.ndarray) -> np.ndarray:
    order = np.argsort(x, kind="mergesort")
    result = np.empty(len(x), dtype=float)
    start = 0
    while start < len(x):
        end = start + 1
        while end < len(x) and x[order[end]] == x[order[start]]:
            end += 1
        result[order[start:end]] = 0.5 * (start + end - 1) + 1
        start = end
    return result


def spearman(x: np.ndarray, y: np.ndarray) -> float:
    return corr(ranks(x), ranks(y))


def standardized_slope(x: np.ndarray, y: np.ndarray) -> float:
    sx = float(np.std(x, ddof=1))
    sy = float(np.std(y, ddof=1))
    return corr(x, y) if sx > 0 and sy > 0 else float("nan")


def summary_stats(
    x: np.ndarray,
    net: np.ndarray,
    adverse: np.ndarray,
) -> dict[str, float]:
    return {
        "net_spearman": spearman(x, net),
        "adverse_spearman": spearman(x, adverse),
        "net_standardized_slope": standardized_slope(x, net),
        "adverse_standardized_slope": standardized_slope(x, adverse),
    }


def tercile_effects(
    x: np.ndarray,
    net: np.ndarray,
    adverse: np.ndarray,
) -> dict[str, float | int]:
    lower_cut = float(np.quantile(x, 1 / 3))
    upper_cut = float(np.quantile(x, 2 / 3))
    lower = x <= lower_cut
    upper = x >= upper_cut
    return {
        "lower_cut": lower_cut,
        "upper_cut": upper_cut,
        "lower_count": int(lower.sum()),
        "upper_count": int(upper.sum()),
        "net_upper_minus_lower": float(net[upper].mean() - net[lower].mean()),
        "adverse_upper_minus_lower": float(
            adverse[upper].mean() - adverse[lower].mean()
        ),
        "lower_net_mean": float(net[lower].mean()),
        "upper_net_mean": float(net[upper].mean()),
        "lower_adverse_mean": float(adverse[lower].mean()),
        "upper_adverse_mean": float(adverse[upper].mean()),
    }


def bootstrap(
    x: np.ndarray,
    net: np.ndarray,
    adverse: np.ndarray,
) -> dict[str, Any]:
    n = len(x)
    if n < BLOCK:
        raise ValueError("insufficient observations for moving-block bootstrap")
    block_count = math.ceil(n / BLOCK)
    rng = np.random.default_rng(SEED)
    names = tuple(summary_stats(x, net, adverse))
    draws = {name: np.empty(DRAWS, dtype=float) for name in names}
    for draw in range(DRAWS):
        starts = rng.integers(0, n - BLOCK + 1, size=block_count)
        indices = (
            starts[:, None] + np.arange(BLOCK, dtype=int)[None, :]
        ).reshape(-1)[:n]
        record = summary_stats(x[indices], net[indices], adverse[indices])
        for name in names:
            draws[name][draw] = record[name]
    result: dict[str, Any] = {}
    for name in names:
        values = draws[name]
        if not np.isfinite(values).all():
            raise ValueError(f"non-finite bootstrap values for {name}")
        quantiles = np.percentile(values, [2.5, 50.0, 97.5])
        result[name] = {
            "lower_95": float(quantiles[0]),
            "median": float(quantiles[1]),
            "upper_95": float(quantiles[2]),
            "nonpositive_mass": float(np.mean(values <= 0)),
        }
    return {
        "draws": DRAWS,
        "block_opportunities": BLOCK,
        "seed": SEED,
        "paired_non_circular_common_index": True,
        "statistics": result,
    }


def candle_rejection_balance(series: Any, end: int) -> tuple[np.ndarray, dict[str, int]]:
    opens = np.asarray(series.opens[:end], dtype=float)
    highs = np.asarray(series.highs[:end], dtype=float)
    lows = np.asarray(series.lows[:end], dtype=float)
    closes = np.asarray(series.closes[:end], dtype=float)
    finite_positive = (
        np.isfinite(opens)
        & np.isfinite(highs)
        & np.isfinite(lows)
        & np.isfinite(closes)
        & (opens > 0)
        & (highs > 0)
        & (lows > 0)
        & (closes > 0)
    )
    geometry_ok = (
        (highs >= np.maximum(opens, closes))
        & (lows <= np.minimum(opens, closes))
        & (highs >= lows)
    )
    violations = np.flatnonzero(~finite_positive | ~geometry_ok)
    if len(violations):
        first = int(violations[0])
        raise ValueError(
            "invalid finite-positive OHLC geometry at index "
            f"{first}: o={opens[first]}, h={highs[first]}, "
            f"l={lows[first]}, c={closes[first]}"
        )
    ranges = highs - lows
    zero_range = ranges == 0
    balance = np.full(end, np.nan, dtype=float)
    valid = ~zero_range
    lower_wick = np.minimum(opens[valid], closes[valid]) - lows[valid]
    upper_wick = highs[valid] - np.maximum(opens[valid], closes[valid])
    balance[valid] = (lower_wick - upper_wick) / ranges[valid]
    if np.any(balance[valid] < -1.000000000001) or np.any(
        balance[valid] > 1.000000000001
    ):
        raise ValueError("rejection balance escaped [-1, 1]")
    return balance, {
        "rows_checked": int(end),
        "ohlc_geometry_violations": 0,
        "zero_range_candles": int(zero_range.sum()),
        "valid_rejection_candles": int(valid.sum()),
    }


def rejection_feature(
    balance: np.ndarray,
    t: int,
) -> dict[str, float | int | bool]:
    baseline = balance[t - 912 : t - 192]
    recent = balance[t - 192 : t - 24]
    if len(baseline) != BASELINE_HOURS or len(recent) != RECENT_HOURS:
        raise ValueError("frozen candle-rejection windows have incorrect length")
    baseline_valid = np.isfinite(baseline)
    recent_valid = np.isfinite(recent)
    baseline_count = int(baseline_valid.sum())
    recent_count = int(recent_valid.sum())
    baseline_fraction = baseline_count / BASELINE_HOURS
    recent_fraction = recent_count / RECENT_HOURS
    valid = (
        baseline_fraction >= MIN_VALID_FRACTION
        and recent_fraction >= MIN_VALID_FRACTION
    )
    if not valid:
        return {
            "valid": False,
            "baseline_valid_candles": baseline_count,
            "recent_valid_candles": recent_count,
            "baseline_valid_fraction": baseline_fraction,
            "recent_valid_fraction": recent_fraction,
            "baseline_balance": float("nan"),
            "recent_balance": float("nan"),
            "candle_rejection_persistence": float("nan"),
        }
    baseline_mean = float(np.mean(baseline[baseline_valid]))
    recent_mean = float(np.mean(recent[recent_valid]))
    return {
        "valid": True,
        "baseline_valid_candles": baseline_count,
        "recent_valid_candles": recent_count,
        "baseline_valid_fraction": baseline_fraction,
        "recent_valid_fraction": recent_fraction,
        "baseline_balance": baseline_mean,
        "recent_balance": recent_mean,
        "candle_rejection_persistence": recent_mean - baseline_mean,
    }


def first_daily_anchor() -> int:
    minimum = 2185
    return ((minimum + 23) // 24) * 24


def opportunities(series: Any, end: int) -> tuple[list[dict[str, Any]], dict[str, int]]:
    balance, geometry_audit = candle_rejection_balance(series, end)
    records: list[dict[str, Any]] = []
    invalid_feature_anchors = 0
    ineligible_e2160 = 0
    evaluated_daily_anchors = 0
    for t in range(first_daily_anchor(), TRAIN_END - 25, 24):
        evaluated_daily_anchors += 1
        own_e2160_positive = series.closes[t - 25] > series.closes[t - 2185]
        if not own_e2160_positive:
            ineligible_e2160 += 1
            continue
        feature = rejection_feature(balance, t)
        if not feature["valid"]:
            invalid_feature_anchors += 1
            continue
        entry = float(series.opens[t])
        exit_price = float(series.opens[t + 24])
        delayed_entry = float(series.opens[t + 1])
        delayed_exit = float(series.opens[t + 25])
        path = np.asarray(series.opens[t : t + 25], dtype=float) / entry - 1.0
        delayed_path = (
            np.asarray(series.opens[t + 1 : t + 26], dtype=float)
            / delayed_entry
            - 1.0
        )
        records.append(
            {
                "decision_index": t,
                "execution_open_ms": int(series.open_ms[t]),
                "e2160_margin": float(
                    math.log(series.closes[t - 25] / series.closes[t - 2185])
                ),
                "candle_rejection_persistence": float(
                    feature["candle_rejection_persistence"]
                ),
                "baseline_balance": float(feature["baseline_balance"]),
                "recent_balance": float(feature["recent_balance"]),
                "baseline_valid_candles": int(feature["baseline_valid_candles"]),
                "recent_valid_candles": int(feature["recent_valid_candles"]),
                "baseline_valid_fraction": float(
                    feature["baseline_valid_fraction"]
                ),
                "recent_valid_fraction": float(feature["recent_valid_fraction"]),
                "net_24h": float(exit_price / entry - 1.0 - ROUND_TRIP_FEE),
                "adverse_24h": float(np.min(path)),
                "delayed_net_24h": float(
                    delayed_exit / delayed_entry - 1.0 - ROUND_TRIP_FEE
                ),
                "delayed_adverse_24h": float(np.min(delayed_path)),
            }
        )
    audit = {
        **geometry_audit,
        "evaluated_daily_anchors": evaluated_daily_anchors,
        "invalid_feature_anchors": invalid_feature_anchors,
        "ineligible_e2160_anchors": ineligible_e2160,
    }
    return records, audit


def fold_audit(records: list[dict[str, Any]]) -> dict[str, Any]:
    folds: list[dict[str, Any]] = []
    for number in range(4):
        start = WARMUP + number * 2160
        end = start + 2160
        subset = [
            record
            for record in records
            if start <= record["decision_index"] < end
        ]
        if len(subset) >= 3:
            x = np.array(
                [record["candle_rejection_persistence"] for record in subset]
            )
            net = np.array([record["net_24h"] for record in subset])
            adverse = np.array([record["adverse_24h"] for record in subset])
            net_slope = standardized_slope(x, net)
            adverse_slope = standardized_slope(x, adverse)
        else:
            net_slope = None
            adverse_slope = None
        folds.append(
            {
                "fold": number + 1,
                "start_index": start,
                "end_index": end,
                "opportunities": len(subset),
                "net_standardized_slope": net_slope,
                "adverse_standardized_slope": adverse_slope,
            }
        )
    positive_net = [
        float(fold["net_standardized_slope"])
        for fold in folds
        if fold["net_standardized_slope"] is not None
        and float(fold["net_standardized_slope"]) > 0
    ]
    return {
        "folds": folds,
        "positive_net_slope_folds": len(positive_net),
        "positive_adverse_slope_folds": sum(
            fold["adverse_standardized_slope"] is not None
            and float(fold["adverse_standardized_slope"]) > 0
            for fold in folds
        ),
        "largest_positive_net_slope_share": (
            max(positive_net) / sum(positive_net) if positive_net else None
        ),
    }


def truncated_series(base: Any, series: Any, end: int) -> Any:
    return base.Series(
        inst_id=series.inst_id,
        open_ms=series.open_ms[:end],
        opens=series.opens[:end],
        highs=series.highs[:end],
        lows=series.lows[:end],
        closes=series.closes[:end],
        normalized_sha256="training-prefix-only",
        manifest=(),
        instrument_evidence={},
    )


def source_prefix_projection(series: Any, end: int) -> list[dict[str, float | int]]:
    return [
        {
            "open_ms": int(series.open_ms[index]),
            "open": float(series.opens[index]),
            "high": float(series.highs[index]),
            "low": float(series.lows[index]),
            "close": float(series.closes[index]),
        }
        for index in range(end)
    ]


def analyse(base: Any, series: Any, target: str) -> dict[str, Any]:
    records, candle_audit = opportunities(series, FULL_END)
    prefix_series = truncated_series(base, series, TRAIN_END)
    prefix_records, prefix_candle_audit = opportunities(prefix_series, TRAIN_END)
    event_hash = base.sha256_bytes(base.canonical_bytes(records))
    prefix_event_hash = base.sha256_bytes(base.canonical_bytes(prefix_records))
    feature_projection = [
        {
            "decision_index": record["decision_index"],
            "candle_rejection_persistence": record[
                "candle_rejection_persistence"
            ],
            "baseline_balance": record["baseline_balance"],
            "recent_balance": record["recent_balance"],
            "baseline_valid_candles": record["baseline_valid_candles"],
            "recent_valid_candles": record["recent_valid_candles"],
        }
        for record in records
    ]
    prefix_feature_projection = [
        {
            "decision_index": record["decision_index"],
            "candle_rejection_persistence": record[
                "candle_rejection_persistence"
            ],
            "baseline_balance": record["baseline_balance"],
            "recent_balance": record["recent_balance"],
            "baseline_valid_candles": record["baseline_valid_candles"],
            "recent_valid_candles": record["recent_valid_candles"],
        }
        for record in prefix_records
    ]
    feature_hash = base.sha256_bytes(base.canonical_bytes(feature_projection))
    prefix_feature_hash = base.sha256_bytes(
        base.canonical_bytes(prefix_feature_projection)
    )
    source_prefix_hash = base.sha256_bytes(
        base.canonical_bytes(source_prefix_projection(series, TRAIN_END))
    )
    truncated_source_prefix_hash = base.sha256_bytes(
        base.canonical_bytes(source_prefix_projection(prefix_series, TRAIN_END))
    )
    x = np.array(
        [record["candle_rejection_persistence"] for record in records]
    )
    net = np.array([record["net_24h"] for record in records])
    adverse = np.array([record["adverse_24h"] for record in records])
    delayed_net = np.array([record["delayed_net_24h"] for record in records])
    delayed_adverse = np.array(
        [record["delayed_adverse_24h"] for record in records]
    )
    values, counts = np.unique(x, return_counts=True)
    distribution = {
        "minimum": float(x.min()),
        "q25": float(np.quantile(x, 0.25)),
        "median": float(np.quantile(x, 0.5)),
        "q75": float(np.quantile(x, 0.75)),
        "maximum": float(x.max()),
        "iqr": float(np.quantile(x, 0.75) - np.quantile(x, 0.25)),
        "distinct_values": int(len(values)),
        "value_counts": {
            f"{value:.12g}": int(count)
            for value, count in zip(values, counts, strict=True)
        },
    }
    associations = summary_stats(x, net, adverse)
    effects = tercile_effects(x, net, adverse)
    uncertainty = bootstrap(x, net, adverse)
    folds = fold_audit(records)
    delayed = {
        "associations": summary_stats(x, delayed_net, delayed_adverse),
        "tercile_effects": tercile_effects(x, delayed_net, delayed_adverse),
    }
    lower_bounds = uncertainty["statistics"]
    delayed_associations = delayed["associations"]
    delayed_effects = delayed["tercile_effects"]
    gates = {
        "at_least_120_opportunities": len(records) >= 120,
        "all_scored_windows_at_least_95_percent_valid": all(
            record["baseline_valid_fraction"] >= MIN_VALID_FRACTION
            and record["recent_valid_fraction"] >= MIN_VALID_FRACTION
            for record in records
        ),
        "at_least_30_distinct_values": distribution["distinct_values"] >= 30,
        "strictly_positive_iqr": distribution["iqr"] > 0,
        "terciles_each_at_least_35": (
            effects["lower_count"] >= 35 and effects["upper_count"] >= 35
        ),
        "net_spearman_positive": associations["net_spearman"] > 0,
        "adverse_spearman_positive": associations["adverse_spearman"] > 0,
        "net_slope_positive": associations["net_standardized_slope"] > 0,
        "adverse_slope_positive": (
            associations["adverse_standardized_slope"] > 0
        ),
        "net_tercile_effect_positive": effects["net_upper_minus_lower"] > 0,
        "adverse_tercile_effect_positive": (
            effects["adverse_upper_minus_lower"] > 0
        ),
        "net_spearman_lower_95_positive": (
            lower_bounds["net_spearman"]["lower_95"] > 0
        ),
        "adverse_spearman_lower_95_positive": (
            lower_bounds["adverse_spearman"]["lower_95"] > 0
        ),
        "net_slope_lower_95_positive": (
            lower_bounds["net_standardized_slope"]["lower_95"] > 0
        ),
        "adverse_slope_lower_95_positive": (
            lower_bounds["adverse_standardized_slope"]["lower_95"] > 0
        ),
        "at_least_3_of_4_positive_net_folds": (
            folds["positive_net_slope_folds"] >= 3
        ),
        "at_least_3_of_4_positive_adverse_folds": (
            folds["positive_adverse_slope_folds"] >= 3
        ),
        "largest_positive_net_fold_share_at_most_60_percent": (
            folds["largest_positive_net_slope_share"] is not None
            and folds["largest_positive_net_slope_share"] <= 0.60
        ),
        "delayed_net_spearman_positive": (
            delayed_associations["net_spearman"] > 0
        ),
        "delayed_adverse_spearman_positive": (
            delayed_associations["adverse_spearman"] > 0
        ),
        "delayed_net_slope_positive": (
            delayed_associations["net_standardized_slope"] > 0
        ),
        "delayed_adverse_slope_positive": (
            delayed_associations["adverse_standardized_slope"] > 0
        ),
        "delayed_net_tercile_effect_positive": (
            delayed_effects["net_upper_minus_lower"] > 0
        ),
        "delayed_adverse_tercile_effect_positive": (
            delayed_effects["adverse_upper_minus_lower"] > 0
        ),
        "source_prefix_invariant": (
            source_prefix_hash == truncated_source_prefix_hash
        ),
        "feature_prefix_invariant": feature_hash == prefix_feature_hash,
        "opportunity_prefix_invariant": event_hash == prefix_event_hash,
        "candle_audit_prefix_invariant": (
            candle_audit["ohlc_geometry_violations"]
            == prefix_candle_audit["ohlc_geometry_violations"]
            and candle_audit["invalid_feature_anchors"]
            == prefix_candle_audit["invalid_feature_anchors"]
            and candle_audit["ineligible_e2160_anchors"]
            == prefix_candle_audit["ineligible_e2160_anchors"]
        ),
    }
    return {
        "target": target,
        "opportunity_count": len(records),
        "first_opportunity": base.utc_iso(records[0]["execution_open_ms"]),
        "last_opportunity": base.utc_iso(records[-1]["execution_open_ms"]),
        "candle_audit": candle_audit,
        "training_prefix_candle_audit": prefix_candle_audit,
        "distribution": distribution,
        "associations": associations,
        "tercile_effects": effects,
        "bootstrap": uncertainty,
        "fold_audit": folds,
        "one_hour_execution_delay": delayed,
        "source_prefix_sha256": source_prefix_hash,
        "truncated_source_prefix_sha256": truncated_source_prefix_hash,
        "event_sha256": event_hash,
        "prefix_event_sha256": prefix_event_hash,
        "feature_sha256": feature_hash,
        "prefix_feature_sha256": prefix_feature_hash,
        "gates": gates,
        "passes": all(gates.values()),
        "strategy_metrics": {
            "training": None,
            "oos": None,
            "full": None,
            "benchmark_comparison": None,
            "turnover": None,
            "maximum_drawdown": None,
            "edge_per_turnover": None,
            "calendar_year_breadth": None,
        },
    }


def report(evidence: dict[str, Any]) -> str:
    lines = [
        "# Own-price candle-rejection persistence opportunity diagnostic",
        "",
        f"- Family: `{FAMILY}`",
        f"- Exact head: `{evidence['exact_head']}`",
        f"- Source contract: `{evidence['source_contract_passed']}`",
        "- Candidate count / grid: `0 / 0`",
        "- OOS accessed: `False`",
        f"- Verdict: `{evidence['verdict']}`",
        "",
    ]
    if not evidence["source_contract_passed"]:
        lines.extend(
            [
                "## Source rejection",
                "",
                evidence["highest_value_failure"],
                "",
            ]
        )
        return "\n".join(lines)
    lines.extend(
        [
            "## Training-only information",
            "",
            "| Target | Anchors | Distinct | IQR | Net rho | Adverse rho | Net slope | Adverse slope | Net tercile bp | Adverse tercile bp | Net folds | Adverse folds | Gates |",
            "|---|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|",
        ]
    )
    for target in evidence["targets"]:
        associations = target["associations"]
        effects = target["tercile_effects"]
        folds = target["fold_audit"]
        lines.append(
            f"| {target['target']} | {target['opportunity_count']} | "
            f"{target['distribution']['distinct_values']} | "
            f"{target['distribution']['iqr']:.6f} | "
            f"{associations['net_spearman']:+.6f} | "
            f"{associations['adverse_spearman']:+.6f} | "
            f"{associations['net_standardized_slope']:+.6f} | "
            f"{associations['adverse_standardized_slope']:+.6f} | "
            f"{effects['net_upper_minus_lower'] * 10000:+.2f} | "
            f"{effects['adverse_upper_minus_lower'] * 10000:+.2f} | "
            f"{folds['positive_net_slope_folds']}/4 | "
            f"{folds['positive_adverse_slope_folds']}/4 | "
            f"{sum(target['gates'].values())}/{len(target['gates'])} |"
        )
    lines.extend(
        [
            "",
            "## Highest-value failure",
            "",
            evidence["highest_value_failure"],
            "",
            "All strategy train/OOS/full, benchmark, turnover, drawdown, "
            "edge-per-turnover and calendar-year fields remain null because "
            "this zero-candidate diagnostic did not open sealed OOS.",
            "",
        ]
    )
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--base-script", required=True)
    parser.add_argument("--tested-head", required=True)
    parser.add_argument("--output-dir", required=True)
    args = parser.parse_args()
    base = load_base(args.base_script)
    output = Path(args.output_dir)
    source_root = output / "source"
    output.mkdir(parents=True, exist_ok=True)
    series: dict[str, Any] = {}
    failures: dict[str, str] = {}
    for inst_id in TARGETS:
        try:
            acquired = base.acquire_series(inst_id, source_root)
            if acquired.normalized_sha256 != EXPECTED_SOURCE_HASHES[inst_id]:
                raise ValueError(
                    "normalized source hash mismatch: "
                    f"{acquired.normalized_sha256} != "
                    f"{EXPECTED_SOURCE_HASHES[inst_id]}"
                )
            series[inst_id] = acquired
        except Exception as exc:
            failures[inst_id] = f"{type(exc).__name__}: {exc}"
    common_grid = len(series) == len(TARGETS) and np.array_equal(
        series[TARGETS[0]].open_ms,
        series[TARGETS[1]].open_ms,
    )
    if len(series) == len(TARGETS) and not common_grid:
        failures["common_grid"] = "BTC and ETH do not share one exact UTC grid"
    source_passed = len(series) == len(TARGETS) and not failures and common_grid
    source_manifest = {
        "family_id": FAMILY,
        "acquisition_base_sha256": base.sha256_bytes(
            Path(args.base_script).read_bytes()
        ),
        "provider": "OKX public market data",
        "bar": "1H",
        "targets": list(TARGETS),
        "expected_rows_per_series": base.EXPECTED_ROWS,
        "expected_source_sha256": EXPECTED_SOURCE_HASHES,
        "series": {
            key: {
                "rows": len(value.open_ms),
                "normalized_sha256": value.normalized_sha256,
                "instrument_evidence": value.instrument_evidence,
                "pages": list(value.manifest),
            }
            for key, value in series.items()
        },
        "failures": failures,
        "common_grid": common_grid,
        "source_contract_passed": source_passed,
    }
    source_manifest_hash = base.persist_json(
        output / "source_manifest.json",
        source_manifest,
    )
    targets: list[dict[str, Any]] = []
    if source_passed:
        for target in TARGETS:
            try:
                targets.append(analyse(base, series[target], target))
            except Exception as exc:
                failures[f"{target}_analysis"] = f"{type(exc).__name__}: {exc}"
                source_passed = False
                targets = []
                break
    bilateral = bool(targets) and all(target["passes"] for target in targets)
    if source_passed and not bilateral:
        failed_gates = []
        for target in targets:
            names = [
                name for name, passed in target["gates"].items() if not passed
            ]
            failed_gates.append(f"{target['target']}: {', '.join(names)}")
        highest_value_failure = (
            "Bilateral candle-rejection persistence support failed: "
            + "; ".join(failed_gates)
        )
    elif source_passed:
        highest_value_failure = (
            "No information gate failed; only a separately preregistered "
            "candidate architecture may proceed."
        )
    else:
        highest_value_failure = "Source or geometry contract failed: " + "; ".join(
            f"{name}: {message}" for name, message in failures.items()
        )
    evidence = {
        "family_id": FAMILY,
        "exact_head": args.tested_head,
        "repository_main": MAIN,
        "generated_at_utc": datetime.now(UTC).isoformat(),
        "canonical_fee_bps_one_way": 5.0,
        "round_trip_opportunity_fee": ROUND_TRIP_FEE,
        "bar_interval": "1H",
        "candidate_count": 0,
        "parameter_grid_count": 0,
        "source_contract_passed": source_passed,
        "source_failures": failures,
        "source_manifest_sha256": source_manifest_hash,
        "source_series_sha256": {
            key: value.normalized_sha256 for key, value in series.items()
        },
        "targets": targets,
        "sample": {
            "rows_per_series": base.EXPECTED_ROWS,
            "warmup": [0, WARMUP],
            "training_information": [WARMUP, TRAIN_END],
            "sealed_oos": [TRAIN_END, OOS_END],
            "unread_suffix": [OOS_END, FULL_END],
            "folds": 4,
            "fold_hours": 2160,
        },
        "training_information_labels_accessed": source_passed,
        "strategy_performance_accessed": False,
        "oos_accessed": False,
        "full_sample_accessed": False,
        "bilateral_training_support_passed": bilateral,
        "targets_passing": sum(target["passes"] for target in targets),
        "highest_value_failure": highest_value_failure,
        "canonical_strategy_changed": False,
        "canonical_mutation_authorized": False,
        "executable_candidate_authorized": False,
        "separate_candidate_predeclaration_authorized": bilateral,
        "prospective_shadow_authorized": False,
        "paper_trading_authorized": False,
        "live_trading_authorized": False,
        "public_data_only": True,
        "synthetic_market_data_used": False,
        "interpolation_or_row_repair_used": False,
        "cross_sectional_ranking_or_selection": False,
        "pairs_or_spreads": False,
        "cointegration_or_statistical_arbitrage": False,
        "market_neutral_or_long_short": False,
        "credentials_used": False,
        "private_endpoints_used": False,
        "accounts_or_orders": False,
        "enabled_adapters": False,
        "leverage_used": False,
        "non_1h_or_15m_input_used": False,
        "verdict": ACCEPT if bilateral else REJECT,
    }
    evidence["machine_readable_verdict"] = {
        key: evidence[key]
        for key in (
            "family_id",
            "candidate_count",
            "parameter_grid_count",
            "source_contract_passed",
            "bilateral_training_support_passed",
            "strategy_performance_accessed",
            "oos_accessed",
            "canonical_strategy_changed",
            "canonical_mutation_authorized",
            "executable_candidate_authorized",
            "separate_candidate_predeclaration_authorized",
            "paper_trading_authorized",
            "live_trading_authorized",
            "verdict",
        )
    }
    evidence_hash = base.persist_json(output / "evidence.json", evidence)
    protocol = {
        "family_id": FAMILY,
        "targets": list(TARGETS),
        "range": "high[h]-low[h]",
        "lower_wick": "min(open[h],close[h])-low[h]",
        "upper_wick": "high[h]-max(open[h],close[h])",
        "rejection_balance": "(lower_wick-upper_wick)/(high-low)",
        "zero_range": "invalid and excluded; no jitter or imputation",
        "ohlc_violation": "source-contract failure; no clipping",
        "baseline_candles": "[t-912,t-193]",
        "recent_candles": "[t-192,t-25]",
        "aggregation": "unweighted finite-observation mean",
        "feature": "mean(recent rejection balance)-mean(baseline rejection balance)",
        "minimum_valid_fraction_each_window": MIN_VALID_FRACTION,
        "own_e2160": "close[t-25] > close[t-2185]",
        "label": "open[t+24]/open[t]-1-0.0010",
        "adverse": "min(open[t:t+25]/open[t]-1), including zero",
        "delay": "same feature; open[t+1] to open[t+25]",
        "candidate_count": 0,
        "parameter_grid_count": 0,
        "bootstrap": {
            "draws": DRAWS,
            "block_opportunities": BLOCK,
            "seed": SEED,
        },
        "oos_access": "prohibited",
    }
    protocol_hash = base.persist_json(output / "protocol.json", protocol)
    report_text = report(evidence)
    (output / "report.md").write_text(report_text, encoding="utf-8")
    hashes = {
        "evidence_sha256": evidence_hash,
        "source_manifest_sha256": source_manifest_hash,
        "protocol_sha256": protocol_hash,
        "report_sha256": base.sha256_bytes(report_text.encode()),
    }
    (output / "hashes.json").write_text(
        json.dumps(hashes, sort_keys=True, indent=2) + "\n",
        encoding="utf-8",
    )
    print(
        json.dumps(
            {
                "family_id": FAMILY,
                "exact_head": args.tested_head,
                "source_contract_passed": source_passed,
                "targets_passing": evidence["targets_passing"],
                "bilateral_training_support_passed": bilateral,
                "verdict": evidence["verdict"],
                "highest_value_failure": highest_value_failure,
                "hashes": hashes,
            },
            sort_keys=True,
        )
    )


if __name__ == "__main__":
    main()

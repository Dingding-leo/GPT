        f"- Exact head: `{evidence['exact_head']}`",
        f"- Fee: exactly `{evidence['canonical_fee_bps_one_way']}` bps one way",
        f"- Candidate count: `{evidence['candidate_count']}`",
        f"- Parameter grid: `{evidence['parameter_grid_count']}`",
        f"- Source contract passed: `{evidence['source_contract_passed']}`",
        f"- Bilateral training support passed: `{evidence['bilateral_training_support_passed']}`",
        f"- OOS accessed: `{evidence['oos_accessed']}`",
        f"- Verdict: `{evidence['verdict']}`",
        "",
    ]
    if not evidence["source_contract_passed"]:
        lines.extend(
            [
                "## Source-contract rejection",
                "",
                f"`{evidence['source_failure']}`",
                "",
                "All economic fields remain null; no target-return path was opened.",
            ]
        )
        return "\n".join(lines) + "\n"
    lines.extend(
        [
            "## Training-only selector support",
            "",
            "| Market | Decisions | Worsening | Vetoes | Quarters | Largest share | Deferred | Difference hours | Pass |",
            "|---|---:|---:|---:|---:|---:|---:|---:|---:|",
        ]
    )
    for market in evidence["markets"]:
        support = market["training_support"]
        lines.append(
            f"| {market['market']} | {support['training_decisions']} | "
            f"{support['worsening_state_rate']:.4%} | {support['entry_vetoes']} | "
            f"{support['distinct_veto_quarters']} | "
            f"{support['largest_veto_quarter_share']} | "
            f"{support['deferred_entries_later_authorised']} | "
            f"{support['candidate_difference_hours']} | {support['passes']} |"
        )
    if not evidence["oos_accessed"]:
        lines.extend(
            [
                "",
                "The bilateral support gate failed before any train, OOS, or full economic path was evaluated.",
            ]
        )
        return "\n".join(lines) + "\n"
    lines.extend(
        [
            "",
            "## Economics",
            "",
            "| Market | OOS candidate return / Sharpe | OOS E2160 return / Sharpe | Always-long return / Sharpe | Turnover candidate / E2160 | Edge/turn candidate / E2160 | Max DD candidate / E2160 | Gates |",
            "|---|---:|---:|---:|---:|---:|---:|---:|",
        ]
    )
    for market in evidence["markets"]:
        oos = market["strategies"]["oos"]
        candidate = oos["candidate"]
        benchmark = oos["e2160"]
        always = oos["always_long"]
        lines.append(
            f"| {market['market']} | {candidate['net_compound_return']:+.4%} / "
            f"{candidate['annualised_hourly_sharpe']} | "
            f"{benchmark['net_compound_return']:+.4%} / "
            f"{benchmark['annualised_hourly_sharpe']} | "
            f"{always['net_compound_return']:+.4%} / "
            f"{always['annualised_hourly_sharpe']} | "
            f"{candidate['one_way_turnover']} / {benchmark['one_way_turnover']} | "
            f"{candidate['edge_per_turnover_bps']} / {benchmark['edge_per_turnover_bps']} | "
            f"{candidate['maximum_drawdown']:+.4%} / {benchmark['maximum_drawdown']:+.4%} | "
            f"{sum(market['gates'].values())}/{len(market['gates'])} |"
        )
    lines.extend(
        [
            "",
            "## Highest-value failure",
            "",
            evidence["highest_value_failure"],
            "",
        ]
    )
    return "\n".join(lines) + "\n"


def identify_failure(markets: list[dict[str, Any]], *, oos_accessed: bool) -> str:
    if not oos_accessed:
        failures: list[str] = []
        for market in markets:
            support = market["training_support"]
            failed = [name for name, passed in support["gates"].items() if not passed]
            failures.append(f"{market['market']}: {', '.join(failed) or 'none'}")
        return "Training-only information support failed: " + "; ".join(failures)
    failures = []
    for market in markets:
        failed = [name for name, passed in market["gates"].items() if not passed]
        failures.append(f"{market['market']}: {', '.join(failed) or 'none'}")
    return "Frozen economic/robustness gates failed: " + "; ".join(failures)

    args = parser.parse_args()

        raise ValueError("tested head must be a lowercase 40-character SHA")
    output = Path(args.output_dir)
    source_root = output / "source"
    output.mkdir(parents=True, exist_ok=True)

    series_by_id: dict[str, Series] = {}
    source_failure: str | None = None
    try:

            series_by_id[inst_id] = acquire_series(inst_id, source_root)
        common_grid = all(
            np.array_equal(series_by_id[SERIES_IDS[0]].open_ms, series_by_id[inst_id].open_ms)
            for inst_id in SERIES_IDS[1:]
        )
        if not common_grid:
            raise SourceContractError("the three public SPOT series do not share one exact grid")
        if any(len(series_by_id[inst_id].open_ms) != EXPECTED_ROWS for inst_id in SERIES_IDS):
            raise SourceContractError("one public SPOT series has an invalid row count")
    except Exception as exc:
        source_failure = f"{type(exc).__name__}: {exc}"

    source_contract_passed = source_failure is None and len(series_by_id) == 3
    source_manifest = {
        "family_id": FAMILY_ID,

    }
    source_manifest_hash = persist_json(output / "source_manifest.json", source_manifest)
    (output / "source_manifest.sha256").write_text(source_manifest_hash + "\n")

    markets: list[dict[str, Any]] = []
    bilateral_support = False
    invariance: dict[str, Any] = {}
    if source_contract_passed:
        exogenous = series_by_id[EXOGENOUS]
        for market in MARKETS:
            spot = series_by_id[market]

            invariance[market] = source_and_feature_invariance(spot, exogenous)
            if not all(invariance[market][key] for key in ("event_prefix_identical", "position_prefix_identical")):
                support["gates"]["future_suffix_and_feature_prefix_invariance"] = False
                support["passes"] = False

            markets.append(null_market_record(market, support))
        bilateral_support = all(market["training_support"]["passes"] for market in markets)

    oos_accessed = False

                    "performance_accessed": True,

                    "passes_all_gates": all(gates.values()),
                }
            )

    accepted = bool(

    )
    verdict = ACCEPT_VERDICT if accepted else REJECT_VERDICT
    highest_value_failure = (

    evidence = {

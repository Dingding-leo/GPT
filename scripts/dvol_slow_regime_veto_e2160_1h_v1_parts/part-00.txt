def digest(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def quarter_label(timestamp: pd.Timestamp) -> str:
    quarter = (timestamp.month - 1) // 3 + 1
    return f"{timestamp.year}-Q{quarter}"


def json_ready(value: Any) -> Any:
    if isinstance(value, dict):
        return {str(key): json_ready(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [json_ready(item) for item in value]
    if isinstance(value, np.ndarray):
        return [json_ready(item) for item in value.tolist()]
    if isinstance(value, (np.integer,)):
        return int(value)
    if isinstance(value, (np.floating,)):
        number = float(value)
        return number if math.isfinite(number) else None
    if isinstance(value, float):
        return value if math.isfinite(value) else None
    if isinstance(value, pd.Timestamp):
        return value.isoformat().replace("+00:00", "Z")
    if isinstance(value, Path):
        return value.as_posix()
    return value


def utc_index(milliseconds: list[int]) -> pd.DatetimeIndex:
    return pd.to_datetime(milliseconds, unit="ms", utc=True)


def persist_json(path: Path, value: Any) -> str:
    payload = (
        json.dumps(json_ready(value), sort_keys=True, indent=2, allow_nan=False).encode()
        + b"\n"
    )
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(payload)
    return digest(payload)


def source_arm(currency: str, output: Path) -> dict[str, Any]:
    market = f"{currency}-USDT"
    dvol_root = output / "source" / "deribit"
    primary = acquire_dvol(currency, DVOL_END_MS, dvol_root, "fixed-primary")
    suffix = acquire_dvol(currency, DVOL_END_MS + HOUR_MS, dvol_root, "future-suffix")
    primary_rows = primary.pop("rows")
    suffix_rows = suffix.pop("rows")
    normalized_hash = digest(canonical(primary_rows))
    expected_hash = EXPECTED_DVOL_SHA256[currency]
    prefix_hash = digest(canonical([row for row in suffix_rows if row[0] <= DVOL_END_MS]))
    suffix_present = any(row[0] == DVOL_END_MS + HOUR_MS for row in suffix_rows)

    snapshot = fetch_okx_one_hour_candles(
        inst_id=market,
        start=SOURCE_START,
        end=PAYOFF_BOUNDARY,
        limit=100,
        pause_seconds=0.12,
        timeout=30.0,
        safety_pages=3,
    )
    paths = write_okx_snapshot(snapshot, output / "source" / "okx" / market)
    candles = snapshot.candles.copy()
    if candles.index.tz is None:
        candles.index = candles.index.tz_localize("UTC")
    else:
        candles.index = candles.index.tz_convert("UTC")
    candles = candles.sort_index()
    signal_candles = candles.loc[SOURCE_START:SIGNAL_END]
    dvol_index = utc_index([int(row[0]) for row in primary_rows])
    target_grid = signal_candles.index
    common_grid = target_grid.equals(dvol_index)
    source_conditions = {
        "dvol_hash_matches_accepted_contract": normalized_hash == expected_hash,
        "dvol_exact_signal_row_count": len(primary_rows) == EXPECTED_SIGNAL_ROWS,
        "dvol_future_suffix_prefix_identity": normalized_hash == prefix_hash and suffix_present,
        "target_exact_row_count_with_payoff_boundary": len(candles) == EXPECTED_TARGET_ROWS,
        "target_exact_boundaries": bool(
            len(candles)
            and candles.index[0] == SOURCE_START
            and candles.index[-1] == PAYOFF_BOUNDARY
        ),
        "target_signal_grid_matches_dvol": common_grid,
        "target_metadata_identity": bool(
            snapshot.metadata.get("instrument_id") == market
            and snapshot.metadata.get("bar") == "1H"
        ),
        "target_pagination_complete": bool(
            snapshot.metadata.get("pagination_complete")
            and snapshot.metadata.get("requested_start_reached")
        ),
        "target_missing_intervals_zero": snapshot.metadata.get("missing_intervals") in (0, None),
        "target_values_finite_positive": bool(
            np.isfinite(candles[["open", "high", "low", "close"]].to_numpy(dtype=float)).all()
            and (candles[["open", "high", "low", "close"]].to_numpy(dtype=float) > 0).all()
        ),
    }
    dvol_close = pd.Series(
        [float(row[4]) for row in primary_rows],
        index=dvol_index,
        name=f"{currency}_dvol_close",
        dtype=float,
    )
    return {
        "currency": currency,
        "market": market,
        "source_contract_passed": all(source_conditions.values()),
        "source_conditions": source_conditions,
        "dvol_normalized_sha256": normalized_hash,
        "dvol_expected_sha256": expected_hash,
        "dvol_fixed_primary": primary,
        "dvol_future_suffix": suffix,
        "okx_normalized_csv_sha256": snapshot.metadata.get("normalized_csv_sha256"),
        "okx_raw_pages_sha256": snapshot.metadata.get("raw_pages_sha256"),
        "okx_source_response_count": snapshot.metadata.get("source_response_count"),
        "okx_source_response_total_bytes": snapshot.metadata.get("source_response_total_bytes"),
        "okx_snapshot_paths": {key: value.as_posix() for key, value in paths.items()},
        "candles": candles,
        "dvol_close": dvol_close,
    }

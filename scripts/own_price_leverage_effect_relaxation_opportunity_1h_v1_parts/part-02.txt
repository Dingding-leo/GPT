                "opportunities": len(subset),
                "net_standardized_slope": net_slope,
                "adverse_standardized_slope": adverse_slope,
            }
        )
    positive_net = [
        float(fold["net_standardized_slope"])
        for fold in folds
        if fold["net_standardized_slope"] is not None and float(fold["net_standardized_slope"]) > 0
    ]
    return {
        "folds": folds,
        "positive_net_slope_folds": len(positive_net),
        "positive_adverse_slope_folds": sum(
            fold["adverse_standardized_slope"] is not None and float(fold["adverse_standardized_slope"]) > 0
            for fold in folds
        ),
        "largest_positive_net_slope_share": max(positive_net) / sum(positive_net) if positive_net else None,
    }


def truncated_series(base: Any, series: Any, end: int) -> Any:
    return base.Series(
        inst_id=series.inst_id,
        open_ms=series.open_ms[:end],
        opens=series.opens[:end],
        highs=series.highs[:end],
        lows=series.lows[:end],
        closes=series.closes[:end],
        normalized_sha256="training-prefix-only",
        manifest=(),
        instrument_evidence={},
    )


def source_prefix_projection(series: Any, end: int) -> list[dict[str, float | int]]:
    return [
        {
            "open_ms": int(series.open_ms[index]),
            "open": float(series.opens[index]),
            "high": float(series.highs[index]),
            "low": float(series.lows[index]),
            "close": float(series.closes[index]),
        }
        for index in range(end)
    ]


def feature_projection(records: list[dict[str, Any]]) -> list[dict[str, Any]]:
    keys = (
        "decision_index",
        "baseline_first_response_index",
        "baseline_last_response_index",
        "recent_first_response_index",
        "recent_last_response_index",
        "baseline_first_antecedent_index",
        "recent_first_antecedent_index",
        "baseline_leverage",
        "recent_leverage",
        "leverage_relaxation",
    )
    return [{key: record[key] for key in keys} for record in records]

    prefix_feature_hash = base.sha256_bytes(base.canonical_bytes(feature_projection(prefix_records)))

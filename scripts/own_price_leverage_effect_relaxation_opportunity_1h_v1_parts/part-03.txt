        "feature_prefix_invariant": feature_hash == prefix_feature_hash,
        "opportunity_prefix_invariant": event_hash == prefix_event_hash,

        "event_sha256": event_hash,
        "prefix_event_sha256": prefix_event_hash,

        },
    }


def report(evidence: dict[str, Any]) -> str:
    lines = [
        "# Own-price leverage-effect relaxation opportunity diagnostic",
        "",
        f"- Family: `{FAMILY}`",
        f"- Exact head: `{evidence['exact_head']}`",
        f"- Source contract: `{evidence['source_contract_passed']}`",
        "- Candidate count / grid: `0 / 0`",
        "- OOS accessed: `False`",
        f"- Verdict: `{evidence['verdict']}`",
        "",
    ]
    if not evidence["source_contract_passed"]:
        lines.extend(["## Source rejection", "", evidence["highest_value_failure"], ""])
        return "\n".join(lines)
    lines.extend(
        [
            "## Training-only information",
            "",
            "| Target | Anchors | Distinct | IQR | Net rho | Adverse rho | Net slope | Adverse slope | Net tercile bp | Adverse tercile bp | Net folds | Adverse folds | Gates |",
            "|---|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|---:|",
        ]
    )
    for target in evidence["targets"]:
        associations = target["associations"]
        effects = target["tercile_effects"]
        folds = target["fold_audit"]
        lines.append(
            f"| {target['target']} | {target['opportunity_count']} | "
            f"{target['distribution']['distinct_values']} | {target['distribution']['iqr']:.8f} | "
            f"{associations['net_spearman']:+.6f} | {associations['adverse_spearman']:+.6f} | "
            f"{associations['net_standardized_slope']:+.6f} | {associations['adverse_standardized_slope']:+.6f} | "
            f"{effects['net_upper_minus_lower'] * 10000:+.2f} | {effects['adverse_upper_minus_lower'] * 10000:+.2f} | "
            f"{folds['positive_net_slope_folds']}/4 | {folds['positive_adverse_slope_folds']}/4 | "
            f"{sum(target['gates'].values())}/{len(target['gates'])} |"
        )
    lines.extend(
        [
            "",
            "## Highest-value failure",
            "",
            evidence["highest_value_failure"],
            "",
            "All strategy train/OOS/full, benchmark, turnover, drawdown, edge-per-turnover and calendar-year fields remain null because this zero-candidate diagnostic did not open sealed OOS.",
            "",
        ]
    )
    return "\n".join(lines)
